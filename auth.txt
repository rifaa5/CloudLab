require("dotenv").config();

const express = require("express");
const { auth, requiresAuth } = require("express-openid-connect");

const app = express();

const config = {
    authRequired: false,
    auth0Logout: true,
    secret: process.env.SECRET,
    baseURL: process.env.BASE_URL,
    clientID: process.env.CLIENT_ID,
    issuerBaseURL: process.env.ISSUER_BASE_URL
};

app.use(auth(config));

app.get("/", (req, res) => {
    const loggedIn = req.oidc.isAuthenticated();

    res.send(`
        <h1>Identity as a Service using Auth0</h1>
        <p>Status: ${loggedIn ? "Logged in" : "Not logged in"}</p>
        <a href="/login">Login</a><br><br>
        <a href="/profile">View Profile</a><br><br>
        <a href="/logout">Logout</a>
    `);
});

app.get("/profile", requiresAuth(), (req, res) => {
    res.send(`
        <h1>Welcome!</h1>
        <p>Name: ${req.oidc.user.name || "Not provided"}</p>
        <p>Email: ${req.oidc.user.email || "Not provided"}</p>
        <a href="/">Home</a>
    `);
});

app.listen(3000, () => {
    console.log("Server running at http://localhost:3000");
});
